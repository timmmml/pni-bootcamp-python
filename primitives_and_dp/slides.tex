%<<<1 ============ PART 2 ============
\section{Part 2 --- Dynamic Programming}
\divider{Part 2}{Dynamic Programming}
\note{Section divider. The framing, stated as a promise: ``Every hard problem
has a solution space that's exponential to enumerate. An algorithmic paradigm is
a disciplined way to avoid enumerating it. Dynamic programming is one paradigm:
exploit substructure so you solve each piece once and reuse it. We'll see it
three times, each adding one new idea, and the third time is the algorithm you'll
use this afternoon.''}

%<<<2 Core idea
\begin{frame}
\mypic{}{
  \node at (50,92)[title]{The core idea: exploit substructure};
  \node[text width=0.8\paperwidth, align=left, anchor=north] at (50,83){
    A problem is a good fit for \textbf{dynamic programming} when it has:
  \begin{enumerate}
    \item \blue{Optimal substructure} --- the solution is built from solutions to \emph{subproblems}.
    \item \blue{Overlapping subproblems} --- the \emph{same} subproblems recur, so caching pays off.
  \end{enumerate}
  };
  \node at (50,50){$\textbf{Dynamic programming} \;=\; \text{recursion} \;+\; \text{memory.}$};
  \greenbox{50}{40}{0.78}{Contrast with \green{divide \& conquer}: there
    subproblems are \emph{disjoint}, never repeat, so no cache is needed. DP is
    D\&C plus a memory --- used when the subproblem graph is a DAG, not a tree.}
}
\note{Put the two conditions up as a checklist they can apply to future
problems. The contrast with divide \& conquer is important: DP is D\&C plus a
memory, used when the subproblem dependency graph is a DAG rather than a tree.
Recognizing substructure in a NEW problem is the actual skill and it's not
automatable --- that's what the three examples train. Bellman coined ``dynamic
programming'' in the 1950s; the RL Bellman equation is the same idea.}
\end{frame}

%<<<2 Two ways
\begin{lrbox}{\boxA}
\begin{lstlisting}
# top-down (memoization)
cache = {}
def f(state):
    if state in cache:
        return cache[state]
    cache[state] = combine(f(...))
    return cache[state]
\end{lstlisting}
\end{lrbox}
\begin{lrbox}{\boxB}
\begin{lstlisting}
# bottom-up (tabulation)
dp = [base, ...]
for state in order:
    dp[state] = combine(dp[...])
return dp[final]
\end{lstlisting}
\end{lrbox}
\begin{frame}
\mypic{}{
  \node at (50,92)[title]{Two ways to fill the memory};
  \node[text width=0.8\paperwidth, align=left, anchor=north] at (50,84){
  \begin{itemize}
    \item \blue{Top-down (memoization):} write the natural recursion, cache each result the first time.
    \item \blue{Bottom-up (tabulation):} identify the dependency order, fill a table from base cases upward.
  \end{itemize}
  Same answers, same complexity; different code shape.};
  \codebox[0.4]{26}{50}{1}{\boxA}
  \codebox[0.4]{74}{50}{1}{\boxB}
}
\note{Both appear in the three examples: Fibonacci shows the duality cleanly,
edit distance is naturally bottom-up (a 2D table), the trellis is bottom-up over
time. Bottom-up is usually faster (no recursion overhead) and gives control over

\begin{frame}
\mypic{}{
  \node at (50,92)[title]{Example 2 --- Edit distance: the recurrence};
  \node[text width=0.8\paperwidth, align=left, anchor=north] at (50,84){
    Let $D[i][j]$ = edit distance between prefixes $a[1{:}i]$ and $b[1{:}j]$.
    Base cases (turn a prefix into the empty string): $\;D[i][0] = i,\;\; D[0][j] = j.$};
  \node at (50,50){$
    D[i][j] =
    \begin{cases}
    D[i-1][j-1], & a_i = b_j \quad(\text{free match})\\[4pt]
    1 + \min\!\begin{cases}
    D[i-1][j] & (\text{delete } a_i)\\
    D[i][j-1] & (\text{insert } b_j)\\
    D[i-1][j-1] & (\text{substitute})
    \end{cases} & a_i \ne b_j
    \end{cases}$};
  \greenbox{50}{22}{0.78}{Answer: $D[m][n]$. Complexity: $O(mn)$ time and space.
    $D[i-1][j-1]$ is needed by three different cells --- that \green{table reuse}
    is the DP payoff.}
}
\note{Walk each branch's meaning. ``Delete a-i'' means we already know how to
align a[1:i-1] with b[1:j], and we pay 1 to drop a-i. Emphasize the overlap:
D[i-1][j-1] is needed by three different cells, so the table reuse is the DP
payoff. O(mn) vs the exponential number of edit sequences.}
\end{frame}

%<<<2 Edit distance code
\begin{lrbox}{\boxA}
\begin{lstlisting}
def edit_distance(a, b):
    m, n = len(a), len(b)
    D = [[0]*(n+1) for _ in range(m+1)]
    for i in range(m+1): D[i][0] = i
    for j in range(n+1): D[0][j] = j
    for i in range(1, m+1):
        for j in range(1, n+1):
            if a[i-1] == b[j-1]:
                D[i][j] = D[i-1][j-1]
            else:
                D[i][j] = 1 + min(D[i-1][j],     # delete
                                  D[i][j-1],     # insert
                                  D[i-1][j-1])   # substitute
    return D[m][n]
\end{lstlisting}
\end{lrbox}

%<<<2 Smoothing posterior
\begin{frame}
\mypic{}{
  \node at (50,92)[title]{Smoothing posterior --- the thing we actually want};
  \node at (50,80){$
    \gamma_t(i) \equiv P(z_t = i \mid x_{1:T}) =
    \frac{\alpha_t(i)\,\beta_t(i)}{\sum_j \alpha_t(j)\,\beta_t(j)}$};
  \node[text width=0.8\paperwidth, align=left, anchor=north] at (50,68){
    ``Given the \textbf{whole} recording, what state was the system in at time $t$?''
  \begin{itemize}
    \item $\alpha_t(i)$ carries evidence from the \blue{past} ($x_{1:t}$),
    \item $\beta_t(i)$ carries evidence from the \blue{future} ($x_{t+1:T}$),
    \item their product, normalized, is the state posterior at each $t$.
  \end{itemize}
  };
  \notebookbox{50}{34}{0.78}{This is what your plot will show:
    $\argmax_i \gamma_t(i)$ overlaid on the true hidden states.}
}
\note{The intuition is the sell: past-evidence times future-evidence = full-data
belief. The denominator is just P(x) (any column of alpha*beta sums to it --- a
nice sanity check). ``Smoothing'' = using all the data to estimate each timepoint
(vs ``filtering,'' which uses only the past). This gamma is the payoff quantity
of the coding session.}
\end{frame}

    Set the trellis costs to negative log-probabilities:};
  \node at (50,68){$
    c_1(j) = -\log \pi_j - \log B_j(x_1), \quad
    c_t(j) = -\log B_j(x_t), \quad
    w(i,j) = -\log A_{ij}.$};
  \node[text width=0.8\paperwidth, align=left, anchor=north] at (50,58){
    Then the min-cost path $V_t(j) = c_t(j) + \min_i[V_{t-1}(i) + w(i,j)]$
    \textbf{is} the most-likely state sequence (Viterbi). Products of
    probabilities become sums of costs; ``most likely'' becomes ``cheapest.''};
  \advancedbox{50}{34}{0.78}{The forward algorithm is the \emph{same} recursion
    with $\min \to \lse$. Working in log space also pre-empts underflow (next slide).}
}

%<<<2 Two gotchas
\begin{frame}
\mypic{}{
  \node at (50,92)[title]{Two gotchas};
  \redbox{50}{85}{0.82}{\textbf{1. Numerical underflow.} Products of many
    probabilities $\to 0$ fast (recall \code{0.1+0.2} from the morning).
  \begin{itemize}
    \item \emph{Pragmatic fix (recommended today):} normalize $\alpha_t$ to sum to
      1 each step (Rabiner scaling); the summed logs of the normalizers give
      $\log P(x_{1:T})$ for free.
    \item \emph{Cleaner fix:} do everything in log space with \code{logsumexp}.
  \end{itemize}
  }
  \redbox{50}{45}{0.82}{\textbf{2. Smoothing $\ne$ decoding.} The pointwise-MAP
    states $\argmax_i \gamma_t(i)$ can differ from the jointly-MAP Viterbi path
    --- the sequence of individually-most-likely states need not be a coherent
    (or even legal) path.}
}
\note{Gotcha 1 is a when-not-if. Recommend scaling for a 35-min build (fewer
lines); mention log-space as the production choice. Gotcha 2 is subtle but
genuine --- one sentence for the strong students; a two-state example where
gamma-argmax picks an impossible transition drives it home in 60 seconds.}
\end{frame}
